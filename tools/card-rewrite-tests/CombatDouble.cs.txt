namespace MegaCrit.Sts2.Core.Commands {
 using MegaCrit.Sts2.Core.CardSelection;
 public static class CardPileCmd {
  public static Task Add(CardModel c, PileType t, CardPilePosition pos = CardPilePosition.Bottom) {
   c.Pile?.Cards.Remove(c); var pile = t.GetPile(c.Owner); c.Pile=pile;
   if(pos==CardPilePosition.Top) pile.Cards.Insert(0,c); else if(pos==CardPilePosition.Random) pile.Cards.Insert(c.Owner.RunState.Rng.CombatCardSelection.Next(pile.Cards.Count+1),c); else pile.Cards.Add(c);
   return Task.CompletedTask;
  }
  public static async Task Add(IEnumerable<CardModel> cards, PileType t, CardPilePosition pos) { foreach(var c in cards.ToArray()) await Add(c,t,pos); }
  public static async Task Draw(PlayerChoiceContext ctx, int n, Player player) { for(int i=0;i<n && PileType.Draw.GetPile(player).Cards.Count>0;i++) await Add(PileType.Draw.GetPile(player).Cards[0],PileType.Hand); }
 }
 public static class CardCmd { public static async Task Exhaust(PlayerChoiceContext c, CardModel card) { card.Owner.Exhausted++; await CardPileCmd.Add(card,PileType.Exhaust); } }
 public static class CardSelectCmd {
  public static readonly List<CardModel[]> Offered = [];
  private static Task<IEnumerable<CardModel>> Select(PlayerChoiceContext ctx, IEnumerable<CardModel> source, CardSelectorPrefs prefs, Func<CardModel,bool>? filter) {
   var eligible=source.Where(c => filter?.Invoke(c) ?? true).ToArray(); Offered.Add(eligible);
   var choice = ctx.Choices.Count>0 ? ctx.Choices.Dequeue() : eligible.Take(prefs.Min).ToArray();
   if(choice.Any(c => !eligible.Contains(c)) || choice.Length>prefs.Max) throw new Exception("Ineligible selection");
   return Task.FromResult<IEnumerable<CardModel>>(choice);
  }
  public static Task<IEnumerable<CardModel>> FromHand(PlayerChoiceContext ctx, Player player, CardSelectorPrefs prefs, Func<CardModel,bool>? filter, CardModel source) => Select(ctx,PileType.Hand.GetPile(player).Cards.Where(c=>c!=source),prefs,filter);
  public static Task<IEnumerable<CardModel>> FromCombatPile(PlayerChoiceContext ctx, CardPile pile, Player player, CardSelectorPrefs prefs, Func<CardModel,bool>? filter=null) => Select(ctx,pile.Cards,prefs,filter);
  public static Task<IEnumerable<CardModel>> FromSimpleGrid(PlayerChoiceContext ctx, IEnumerable<CardModel> cards, Player player, CardSelectorPrefs prefs) => Select(ctx,cards,prefs,null);
 }
 public static class CreatureCmd {
  public static Task GainBlock(Creature owner, BlockVar v, CardPlay? cp) => GainBlock(owner,v.BaseValue,v.Props,cp);
  public static Task GainBlock(Creature owner, decimal n, ValueProp props, CardPlay? cardPlay) { owner.Block+=n; return Task.CompletedTask; }
 }
 public static class PowerCmd {
  public static Task<T?> Apply<T>(PlayerChoiceContext ctx, Creature owner, decimal n, Creature applier, CardModel? source) where T:PowerModel,new() {
   if(!owner.Powers.TryGetValue(typeof(T),out var p)) owner.Powers[typeof(T)]=p=new T{Owner=owner}; p.Amount+=(int)n; return Task.FromResult((T?)p);
  }
  public static Task Remove(PowerModel p) { p.Owner.Powers.Remove(p.GetType()); return Task.CompletedTask; }
  public static async Task ModifyAmount(PlayerChoiceContext c, PowerModel p, decimal n, Creature? a, CardModel? s) { p.Amount+=(int)n; if(p.Amount<=0) await Remove(p); }
 }
 public static class DamageCmd { public static AttackBuilder Attack(decimal n) => new(n); }
 public sealed class AttackBuilder(decimal damage) {
  private int hits=1; private Creature[] targets=[];
  public AttackBuilder WithHitCount(int n) { hits=n; return this; }
  public AttackBuilder FromCard(CardModel card, CardPlay? play) => this;
  public AttackBuilder Targeting(Creature c) { targets=[c]; return this; }
  public AttackBuilder TargetingAllOpponents(CombatState s) { targets=s.Enemies.ToArray(); return this; }
  public Task Execute(PlayerChoiceContext ctx) { foreach(var t in targets) for(int i=0;i<hits && t.IsAlive;i++) { t.Hp-=(int)damage; if(!t.IsAlive) t.Powers.Clear(); } return Task.CompletedTask; }
 }
